# ============================================================
# MASTER MACHINE (EC2-1)
# Login to the Master EC2 Instance Terminal
# ============================================================

sudo apt update && sudo apt install -y ansible && ansible --version

ansible localhost -m ping


# ============================================================
# MASTER MACHINE (EC2-1)
# Generate SSH Key Pair
# ============================================================

ssh-keygen

# Press Enter for all prompts

cat ~/.ssh/id_ed25519.pub

# Copy the complete output shown


# ============================================================
# NODE MACHINE (EC2-2)
# Login to the Node EC2 Instance Terminal
# ============================================================

echo "<MASTER_PUBLIC_KEY>" >> ~/.ssh/authorized_keys


# ============================================================
# MASTER MACHINE (EC2-1)
# Create Inventory File
# ============================================================

nano ~/inventory.ini

# Paste the following:

[node_machines]
<NODE_PUBLIC_IP> ansible_ssh_user=ubuntu

# Save:
# CTRL + O
# Enter
# CTRL + X


# ============================================================
# MASTER MACHINE (EC2-1)
# Create Playbook File
# ============================================================

nano ~/install_package.yml

# Paste the following:

---
- name: Install a package on the node machine
  hosts: node_machines
  become: yes

  tasks:
    - name: Update apt cache
      apt:
        update_cache: yes

    - name: Install Nginx
      apt:
        name: nginx
        state: present

# Save:
# CTRL + O
# Enter
# CTRL + X


# ============================================================
# MASTER MACHINE (EC2-1)
# Verify SSH Access to Node Machine
# ============================================================

ssh ubuntu@<NODE_PUBLIC_IP>

# If login succeeds, type:

exit


# ============================================================
# MASTER MACHINE (EC2-1)
# Execute Ansible Playbook
# ============================================================

ansible-playbook -i ~/inventory.ini ~/install_package.yml


# ============================================================
# NODE MACHINE (EC2-2)
# Verify Nginx Installation
# ============================================================

systemctl status nginx


# ============================================================
# AWS CONSOLE
# Open Node Machine Security Group
# ============================================================

# EC2 → Instances → Select Node Machine

# Security → Security Groups → Edit Inbound Rules

# Add Rule:
# Type: HTTP
# Port: 80
# Source: Anywhere IPv4 (0.0.0.0/0)

# Save Rules


# ============================================================
# BROWSER
# ============================================================

# Open:

http://<NODE_PUBLIC_IP>

# Welcome to nginx!
